ADO.Net Core:
Backend Application		:	Where we can maintain the data 	=> 	SQL Server(DBMS)
Frontend Application	:	Where Enduser can interact 		=>	.Net App, Java App, Angular App, React JS etc.
		- Client Side Aplication
			- HTML, CSS, Javascript, Angular, React, Express JS etc.
		- Server Side Application
			- .Net(ASP.Net Core, Web API, MicroServices), Java(Spring Boot, JSP, API, Microservice), Phython etc.
Middleware				:	These are used to provide the access to frontend and backend applications.
		- ADO.Net Core		=>	ORM(Object Relational Mapping)	=>	Entity Framework Core, Dapper		
			
Console App(C#.Net) 		<=  ADO.Net Core  =>			SqlServer(T-SQL)

- ADO.Net Core is available in a library format.
- It is a collection of classes, by using these classes we can connect with DB and perform the CRUD operations
- By using ADO.Net Core we are able to connect with DB in two ways
	- Connection Oriented Architecture (Live)
		- Classes:	SqlConnection, SqlCommand, SqlDataReader etc.
		- Stays to open the connection while reading/performing the operation.
		- Fast forward only, it reads only single record at a time.		
		Steps:
		- Prepare the connection 
		- Prepare and send the query/stored procedure along with connection object (SqlCommand)
		- Open the connection		
		- Execute the Query
			- ExecuteNonQuery()	:	Action Queries (Insert/Update/Delete) - int
			- ExecuteReader()	:	Non-Action Query (Read) - SqlDataReader
			- ExecuteScalar()	:	Non-Action Query with agregate function result - object
		- Close the connection
		
	- Disconnection Oriented Architecture (Cached)
		- Classes: SqlDataAdapter, DataSet/DataTable etc.
		- Opens, fills data and closes the connection.
		- Working with data offline, binding to grids and batch updates.
		Steps:
		- Prepare the connection 
		- Prepare and send the select query along with connection object (SqlDataAdapter) 
		- Prepare the Dataset/Table at frontend.
		- Fill the Data
		- Perform the operations
		- Update all data back to server.
		Note: A Dataset is a collection of DataTables, and a DataTable is a collection of DataRow.

END

-- ============================================================
-- BANK ACCOUNT MANAGEMENT SYSTEM - Database Schema
-- ============================================================

CREATE DATABASE BankDB;
GO
USE BankDB;
GO

-- ------------------------------------------------------------
-- TABLE: Accounts
-- ------------------------------------------------------------
CREATE TABLE Accounts
(
    AccNo         INT IDENTITY(1001,1) PRIMARY KEY,
    CustomerName  VARCHAR(100) NOT NULL,
    AccType       VARCHAR(20)  NOT NULL,   -- Savings / Current
    Balance       DECIMAL(12,2) NOT NULL DEFAULT 0,
    CreatedOn     DATETIME DEFAULT GETDATE()
);
GO

-- ------------------------------------------------------------
-- TABLE: TransactionLog
-- ------------------------------------------------------------
CREATE TABLE TransactionLog
(
    TransId    INT IDENTITY(1,1) PRIMARY KEY,
    AccNo      INT NOT NULL FOREIGN KEY REFERENCES Accounts(AccNo),
    TransType  VARCHAR(20) NOT NULL,       -- Deposit / Withdraw / Transfer-Out / Transfer-In
    Amount     DECIMAL(12,2) NOT NULL,
    TransDate  DATETIME DEFAULT GETDATE()
);
GO

-- ============================================================
-- STORED PROCEDURE: sp_CreateAccount
-- Inserts a new account, returns the generated AccNo via OUTPUT
-- ============================================================
CREATE PROCEDURE sp_CreateAccount
    @CustomerName VARCHAR(100),
    @AccType      VARCHAR(20),
    @Balance      DECIMAL(12,2),
    @NewAccNo     INT OUTPUT
AS
BEGIN
    INSERT INTO Accounts (CustomerName, AccType, Balance)
    VALUES (@CustomerName, @AccType, @Balance);

    SET @NewAccNo = SCOPE_IDENTITY();
END
GO

-- ============================================================
-- STORED PROCEDURE: sp_Deposit
-- Adds amount to balance + logs the transaction
-- ============================================================
CREATE PROCEDURE sp_Deposit
    @AccNo  INT,
    @Amount DECIMAL(12,2)
AS
BEGIN
    IF NOT EXISTS (SELECT 1 FROM Accounts WHERE AccNo = @AccNo)
    BEGIN
        RAISERROR('Account not found.', 16, 1);
        RETURN;
    END

    UPDATE Accounts SET Balance = Balance + @Amount WHERE AccNo = @AccNo;
    INSERT INTO TransactionLog (AccNo, TransType, Amount) VALUES (@AccNo, 'Deposit', @Amount);
END
GO

-- ============================================================
-- STORED PROCEDURE: sp_Withdraw
-- Deducts amount only if sufficient balance exists
-- ============================================================
CREATE PROCEDURE sp_Withdraw
    @AccNo  INT,
    @Amount DECIMAL(12,2)
AS
BEGIN
    DECLARE @CurrentBalance DECIMAL(12,2);

    SELECT @CurrentBalance = Balance FROM Accounts WHERE AccNo = @AccNo;

    IF @CurrentBalance IS NULL
    BEGIN
        RAISERROR('Account not found.', 16, 1);
        RETURN;
    END

    IF @CurrentBalance < @Amount
    BEGIN
        RAISERROR('Insufficient balance.', 16, 1);
        RETURN;
    END

    UPDATE Accounts SET Balance = Balance - @Amount WHERE AccNo = @AccNo;
    INSERT INTO TransactionLog (AccNo, TransType, Amount) VALUES (@AccNo, 'Withdraw', @Amount);
END
GO

-- ============================================================
-- STORED PROCEDURE: sp_TransferAmount
-- Transfers between two accounts inside a SQL transaction
-- (rolls back both updates if anything fails midway)
-- ============================================================
CREATE PROCEDURE sp_TransferAmount
    @FromAccNo INT,
    @ToAccNo   INT,
    @Amount    DECIMAL(12,2)
AS
BEGIN
    BEGIN TRANSACTION;

    BEGIN TRY
        DECLARE @FromBalance DECIMAL(12,2);
        SELECT @FromBalance = Balance FROM Accounts WHERE AccNo = @FromAccNo;

        IF @FromBalance IS NULL
            THROW 50001, 'Source account not found.', 1;

        IF @FromBalance < @Amount
            THROW 50002, 'Insufficient balance in source account.', 1;

        IF NOT EXISTS (SELECT 1 FROM Accounts WHERE AccNo = @ToAccNo)
            THROW 50003, 'Destination account not found.', 1;

        UPDATE Accounts SET Balance = Balance - @Amount WHERE AccNo = @FromAccNo;
        UPDATE Accounts SET Balance = Balance + @Amount WHERE AccNo = @ToAccNo;

        INSERT INTO TransactionLog (AccNo, TransType, Amount) VALUES (@FromAccNo, 'Transfer-Out', @Amount);
        INSERT INTO TransactionLog (AccNo, TransType, Amount) VALUES (@ToAccNo, 'Transfer-In', @Amount);

        COMMIT TRANSACTION;
    END TRY
    BEGIN CATCH
        ROLLBACK TRANSACTION;
        THROW;   -- re-throws the original error to the caller (ADO.NET will catch it as SqlException)
    END CATCH
END
GO

-- ============================================================
-- STORED PROCEDURE: sp_GetBalance
-- Returns balance via OUTPUT parameter (no result set)
-- ============================================================
CREATE PROCEDURE sp_GetBalance
    @AccNo   INT,
    @Balance DECIMAL(12,2) OUTPUT
AS
BEGIN
    SELECT @Balance = Balance FROM Accounts WHERE AccNo = @AccNo;
END
GO

-- ============================================================
-- STORED PROCEDURE: sp_GetAccountByNo
-- ============================================================
CREATE PROCEDURE sp_GetAccountByNo
    @AccNo INT
AS
BEGIN
    SELECT AccNo, CustomerName, AccType, Balance, CreatedOn
    FROM Accounts
    WHERE AccNo = @AccNo;
END
GO

-- ============================================================
-- STORED PROCEDURE: sp_GetAllAccounts
-- ============================================================
CREATE PROCEDURE sp_GetAllAccounts
AS
BEGIN
    SELECT AccNo, CustomerName, AccType, Balance, CreatedOn FROM Accounts;
END
GO

-- ============================================================
-- STORED PROCEDURE: sp_GetTransactionHistory
-- ============================================================
CREATE PROCEDURE sp_GetTransactionHistory
    @AccNo INT
AS
BEGIN
    SELECT TransId, AccNo, TransType, Amount, TransDate
    FROM TransactionLog
    WHERE AccNo = @AccNo
    ORDER BY TransDate DESC;
END
GO

Data Access Layer(DAL) - Talks to the database only
